Ubuntu Touch SSH bootstrap

mkdir -p ~/.ssh && chmod 700 ~/.ssh
wget -qO- https://portal.3dvr.tech/api/ut-pair >> ~/.ssh/authorized_keys
chmod 600 ~/.ssh/authorized_keys
sudo systemctl start ssh.socket
[ -f ~/.ssh/utunnel ] || ssh-keygen -q -t ed25519 -N '' -f ~/.ssh/utunnel
K="$(base64 ~/.ssh/utunnel.pub | tr -d '\n')"
wget -qO- --post-data="key=$K" https://portal.3dvr.tech/api/ut-pair
